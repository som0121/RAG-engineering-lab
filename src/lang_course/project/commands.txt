# Rate limit

foreach ($n in 1..22) {
    try {
        $r = Invoke-WebRequest -Method Post -Uri "$api/chat" -ContentType "application/json" -Body '{"message":"What is LangGraph? Answer in two sentences."}' -UseBasicParsing
        "request $n -> $($r.StatusCode)"
    } catch {
        "request $n -> $($_.Exception.Response.StatusCode.value__) $($_.ErrorDetails.Message)"
    }
}
